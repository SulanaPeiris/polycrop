// ================= TIME =================
bool initTime() {
  configTime(19800, 0, "pool.ntp.org", "time.nist.gov"); // Sri Lanka UTC+5:30
  time_t now = time(nullptr);
  int retry = 0;

  while (now < 1700000000 && retry < 60) {
    delay(500);
    now = time(nullptr);
    retry++;
  }

  return now >= 1700000000;
}

String nowRFC3339Z() {
  time_t now = time(nullptr);
  if (now < 1700000000) return "1970-01-01T00:00:00Z";

  struct tm tm_utc;
  gmtime_r(&now, &tm_utc);

  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &tm_utc);
  return String(buf);
}

// ================= WIFI =================
void connectWiFi() {
  Serial.print("Connecting WiFi");
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  unsigned long start = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - start < 15000) {
    Serial.print(".");
    delay(500);
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println();
    Serial.println("WiFi connected");
    Serial.print("IP: ");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println();
    Serial.println("WiFi failed");
  }
}

// ================= FIREBASE =================
bool initFirebase() {
  config.api_key = API_KEY;
  config.token_status_callback = tokenStatusCallback;

  if (Firebase.signUp(&config, &auth, "", "")) {
    Serial.println("Firebase anonymous auth OK");
  } else {
    Serial.print("Firebase signup failed: ");
    Serial.println(config.signer.signupError.message.c_str());
  }

  fbdo.setBSSLBufferSize(4096, 1024);
  fbdo.setResponseSize(4096);

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  unsigned long start = millis();
  while (!Firebase.ready() && millis() - start < 8000) {
    delay(200);
  }

  Serial.println(Firebase.ready() ? "Firebase ready" : "Firebase not ready");
  return Firebase.ready();
}

// ================= ULTRASONIC =================
void initUltrasonic() {
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  digitalWrite(TRIG_PIN, LOW);
}

float readDistanceCM() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  unsigned long duration = pulseIn(ECHO_PIN, HIGH, 30000);

  if (duration == 0) return NAN;

  float distance = duration * 0.0343f / 2.0f;

  if (distance < 2.0f || distance > 400.0f) return NAN;

  return distance;
}

float readAverageDistanceCM(int samples = 5) {
  float total = 0;
  int count = 0;

  for (int i = 0; i < samples; i++) {
    float d = readDistanceCM();
    if (!isnan(d)) {
      total += d;
      count++;
    }
    delay(50);
  }

  if (count == 0) return NAN;
  return total / count;
}

// ================= FIRESTORE WRITE =================
bool storeDistanceToFirestore(float distanceCm) {
  if (!Firebase.ready()) return false;

  String docPath = "ultrasonicReadings/reading_latest";

  FirebaseJson content;
  content.set("fields/deviceId/stringValue", String(DEVICE_ID));
  content.set("fields/distanceCm/doubleValue", distanceCm);
  content.set("fields/ts/timestampValue", nowRFC3339Z());

  String updateMask = "deviceId,distanceCm,ts";

  bool ok = Firebase.Firestore.patchDocument(
    &fbdo,
    FIREBASE_PROJECT_ID,
    "",
    docPath.c_str(),
    content.raw(),
    updateMask.c_str()
  );

  if (!ok) {
    Serial.print("Firestore write FAILED: ");
    Serial.println(fbdo.errorReason());
  }

  return ok;
}

// ================= SETUP =================
void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("ESP32 + HC-SR04 + Firebase");

  initUltrasonic();

  connectWiFi();

  Serial.println("Syncing time...");
  bool timeOk = initTime();
  Serial.println(timeOk ? "Time OK" : "Time FAILED");

  initFirebase();
}

// ================= LOOP =================
void loop() {
  Firebase.ready(); // keep token alive

  if (millis() - lastReadMs >= READ_INTERVAL_MS) {
    lastReadMs = millis();

    float distance = readAverageDistanceCM(5);
    lastDistanceCm = distance;

    if (isnan(distance)) {
      Serial.println("Distance: Out of range or no signal");
    } else {
      Serial.print("Distance: ");
      Serial.print(distance, 1);
      Serial.println(" cm");
    }
  }

  if (millis() - lastStoreMs >= STORE_INTERVAL_MS) {
    lastStoreMs = millis();

    if (WiFi.status() != WL_CONNECTED) {
      connectWiFi();
      initTime();
    }

    if (!isnan(lastDistanceCm)) {
      bool ok = storeDistanceToFirestore(lastDistanceCm);
      Serial.println(ok ? "Firestore write OK" : "Firestore write FAILED");
    }
  }
}
