// ================= TIME SYNC (TLS) =================
void initTime() {
  configTime(19800, 0, "pool.ntp.org", "time.nist.gov"); // Sri Lanka UTC+5:30
  Serial.print("Syncing time");
  time_t now = time(nullptr);
  int retry = 0;
  while (now < 1700000000 && retry < 60) {
    Serial.print(".");
    delay(500);
    now = time(nullptr);
    retry++;
  }
  Serial.println();
  if (now < 1700000000) Serial.println("Time sync failed. TLS may fail.");
}

String nowRFC3339Z() {
  time_t now = time(nullptr);
  if (now < 1700000000) return "1970-01-01T00:00:00Z";

  struct tm tm_utc;
  gmtime_r(&now, &tm_utc);

  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &tm_utc);
  return String(buf);
}

// ================= RFID =================
String getRFIDTag() {
  String tagID = "";
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    tagID += String(mfrc522.uid.uidByte[i] < 0x10 ? "0" : "");
    tagID += String(mfrc522.uid.uidByte[i], HEX);
  }
  tagID.toUpperCase();
  return tagID;
}

void fetchAssignedTunnelId() {
  if (!Firebase.ready()) return;

  String path = String(ROBOTS_COLLECTION) + "/" + String(ROBOT_ID);

  if (!Firebase.Firestore.getDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str())) {
    Serial.print("fetchAssignedTunnelId getDocument FAILED: ");
    Serial.println(fbdo.errorReason());
    return;
  }

  FirebaseJson json;
  json.setJsonData(fbdo.payload().c_str());

  FirebaseJsonData data;
  json.get(data, "fields/assignedTunnelId/stringValue");

  if (data.success) {
    String v = data.to<String>();
    if (v != assignedTunnelId) {
      assignedTunnelId = v;
      Serial.print("Assigned tunnel updated: ");
      Serial.println(assignedTunnelId);
    }
  }
}

// Reads: tunnels/{assignedTunnelId}/rfidMap/{RFID} -> fields/plantId/stringValue
void resolvePlantDocIdFromRFID() {
  currentPlantDocId = currentRFID; // fallback (old behavior)
  currentRFIDSide = "";

  if (!Firebase.ready()) return;
  if (assignedTunnelId.length() == 0) return;
  if (currentRFID == "NONE") return;

  String mapPath = String(TUNNELS_COLLECTION) + "/" + assignedTunnelId + "/" +
                   String(RFIDMAP_SUBCOLLECTION) + "/" + currentRFID;

  if (!Firebase.Firestore.getDocument(&fbdo, FIREBASE_PROJECT_ID, "", mapPath.c_str())) {
    return;
  }

  FirebaseJson json;
  json.setJsonData(fbdo.payload().c_str());

  FirebaseJsonData data;

  json.get(data, "fields/plantId/stringValue");
  if (data.success) currentPlantDocId = data.to<String>();

  json.get(data, "fields/side/stringValue");
  if (data.success) currentRFIDSide = data.to<String>();
}

bool fetchDecisionForRequest(const String& reqId, String& outDecision, int& outSprayMs) {
  if (!Firebase.ready()) return false;

  String path = String(ROBOTS_COLLECTION) + "/" + String(ROBOT_ID);

  if (!Firebase.Firestore.getDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str())) {
    return false;
  }

  FirebaseJson json;
  json.setJsonData(fbdo.payload().c_str());

  FirebaseJsonData data;

  json.get(data, "fields/captureRequestId/stringValue");
  if (!data.success) return false;
  String gotReq = data.to<String>();
  if (gotReq != reqId) return false;

  json.get(data, "fields/captureStatus/stringValue");
  if (!data.success) return false;
  String st = data.to<String>();
  if (st != "DECIDED") return false;

  json.get(data, "fields/captureDecision/stringValue");
  outDecision = data.success ? data.to<String>() : "NO_SPRAY";

  json.get(data, "fields/sprayDurationMs/integerValue");
  outSprayMs = data.success ? data.to<int>() : 0;

  return true;
}

void sprayNow(int ms) {
  if (ms <= 0) return;

  actuatorStatus = "SPRAYING";
  pushDataToFirebase();
  logScanEvent("SPRAY_START", currentPositionLabel + " stop#" + String(stepperStopIndex) + " ms=" + String(ms));

  // Pump ON
  pump1On();

  // Optional nozzle wiggle
  nozzleServo.attach(SERVO_PIN, 500, 2400);
  delay(100);

  nozzleServo.write(SERVO_ANGLE_A);
  delay(200);

  if (ms >= 2000) {
    nozzleServo.write(SERVO_ANGLE_B);
    delay(ms - 400);
    nozzleServo.write(SERVO_ANGLE_A);
    delay(200);
  } else {
    delay(ms);
  }

  nozzleServo.detach();
  pump1Off();

  actuatorStatus = "OFF";
  pushDataToFirebase();
  logScanEvent("SPRAY_DONE", currentPositionLabel + " stop#" + String(stepperStopIndex));
}

void finalizePositionAfterScan() {
  String donePos = currentPositionLabel;

  if (nozzleServo.attached()) {
    nozzleServo.detach();
    delay(100);
  }

  mode = ACTUATION;
  modeStr = "ACTUATION";
  scanningStatus = "DONE";
  robotMovingStatus = "STOPPED";
  actuatorStatus = "OFF";
  pushDataToFirebase();

  logScanEvent("STEPPER_SCAN_DONE", donePos + " stepper returned to 0 (position scan done)");
  logScanEvent("POSITION_DONE", donePos + " complete (decision + spray handled per stop)");

  // If A/B/C all complete -> mark plant done and reset plant state
  if (plantStopCount >= 3) {
    updatePlantDocAfterAll();
    logScanEvent("PLANT_DONE", "A/B/C complete. Plant finished.");

    plantStopCount = 0;
    currentPositionLabel = "NONE";
    currentRFID = "NONE";
    currentPlantDocId = "";
    currentRFIDSide = "";
    rfidDetected = false;
    bothBlackLatched = false;
  }

  // Clear capture state
  captureStatus = "IDLE";
  captureRequestId = "";
  captureDecision = "";
  sprayDurationMs = 0;
  captureId = "";
  captureImageUrl = "";

  pushDataToFirebase();

  // Resume line following
  robotMovingStatus = "MOVING";
  scanningStatus = "IDLE";
  actuatorStatus = "OFF";
  mode = LINE_FOLLOW;
  modeStr = "LINE_FOLLOW";
  bothBlackLatched = false;
  pushDataToFirebase();

  logScanEvent("RESUME", "Resuming line follow after " + donePos);
  Serial.println(">>> Moving to next position...");
  moveForwardToClearMarker();
}

void runActuationSequence() {
  mode = ACTUATION;
  modeStr = "ACTUATION";
  scanningStatus = "DONE";
  robotMovingStatus = "STOPPED";
  pushDataToFirebase();

  logScanEvent("STEPPER_SCAN_DONE", currentPositionLabel + " stepper returned to 0, starting actuation");

  // Pump1: 5s
  actuatorStatus = "PUMP1_ON";
  pushDataToFirebase();
  logScanEvent("PUMP1_ON", currentPositionLabel + " pump1 ON for 5s");

  pump1On();
  delay(5000);
  pump1Off();

  actuatorStatus = "OFF";
  pushDataToFirebase();
  logScanEvent("PUMP1_OFF", currentPositionLabel + " pump1 OFF");

  // Pump2: 5s
  actuatorStatus = "PUMP2_ON";
  pushDataToFirebase();
  logScanEvent("PUMP2_ON", currentPositionLabel + " pump2 ON for 5s");

  pump2On();
  delay(5000);
  pump2Off();

  actuatorStatus = "OFF";
  pushDataToFirebase();
  logScanEvent("PUMP2_OFF", currentPositionLabel + " pump2 OFF");

  // Servo
  actuatorStatus = "SERVO_MOVING";
  pushDataToFirebase();
  logScanEvent("SERVO_START", currentPositionLabel + " servo 40deg 5s -> 90deg 5s");

  stopMotors();
  delay(100);

  nozzleServo.attach(SERVO_PIN, 500, 2400);
  delay(100);

  nozzleServo.write(SERVO_ANGLE_A);   // 40 deg
  delay(5000);

  nozzleServo.write(SERVO_ANGLE_B);   // 90 deg
  delay(5000);

  nozzleServo.detach();
  delay(100);

  actuatorStatus = "OFF";
  pushDataToFirebase();
  logScanEvent("SERVO_DONE", currentPositionLabel + " servo sequence done");

  updatePlantDocAfterAll();
  logScanEvent("POSITION_DONE", currentPositionLabel + " complete");

  if (plantStopCount >= 3) {
    Serial.println(">>> Plant cycle complete. Resuming line follow for next plant...");
    logScanEvent("PLANT_CYCLE_DONE", "A/B/C complete. Resuming line follow for next plant");

    plantStopCount = 0;
    currentPositionLabel = "NONE";
    currentRFID = "NONE";
    currentPlantDocId = "";
    rfidDetected = false;
    bothBlackLatched = false;
  }

  // Resume line following
  robotMovingStatus = "MOVING";
  scanningStatus = "IDLE";
  actuatorStatus = "OFF";
  mode = LINE_FOLLOW;
  modeStr = "LINE_FOLLOW";
  bothBlackLatched = false;
  pushDataToFirebase();
  logScanEvent("RESUME", "Resuming line follow after " + currentPositionLabel);

  Serial.println(">>> Moving to next position...");
  moveForwardToClearMarker();
}

  modeStr = "STEPPER_SCAN";
  stopMotors();

  // -------- Running to next stop --------
  if (!stepperPauseActive) {
    stepper.run();

    if (stepper.distanceToGo() == 0) {
      // If we are returning and already at 0 -> scan complete for this position
      if (!movingForward && stepper.currentPosition() == 0 && nextStop == 0) {
        Serial.println("Stepper back at 0. Scan complete.");
        logScanEvent("STEPPER_BACK_AT_ZERO", currentPositionLabel + " stepper returned to 0");
        finalizePositionAfterScan();
        return;
      }

      stepperPauseActive = true;
      captureWaitActive = true;
      captureWaitStartMs = millis();

      stepperStopIndex++;

      long rounds = stepper.currentPosition() / MOTOR_STEPS_PER_REV;
      String directionLabel = movingForward ? "FWD" : "BWD";

      Serial.print("Stop reached at ");
      Serial.print(rounds);
      Serial.print(" rounds. StopIndex=");
      Serial.println(stepperStopIndex);

      logScanEvent("STEPPER_STOP", currentPositionLabel + " stop#" + String(stepperStopIndex) + " rounds=" + String(rounds));

      requestCaptureAtStop((int)rounds, directionLabel);
    }
    return;
  }

  // -------- Paused: waiting for ML decision --------
  if (captureWaitActive) {
    String decision = "";
    int ms = 0;

    bool decided = fetchDecisionForRequest(captureRequestId, decision, ms);

    if (!decided) {
      if (millis() - captureWaitStartMs > CAPTURE_WAIT_TIMEOUT_MS) {
        decision = "NO_SPRAY";
        ms = 0;
        logScanEvent("CAPTURE_TIMEOUT", currentPositionLabel + " stop#" + String(stepperStopIndex) + " timeout, continue without spray");
      } else {
        delay(250); // small yield
        return;
      }
    }

    captureDecision = decision;
    sprayDurationMs = ms;

    // Apply decision
    if (captureDecision == "SPRAY" && sprayDurationMs > 0) {
      sprayNow(sprayDurationMs);
    } else {
      logScanEvent("NO_SPRAY", currentPositionLabel + " stop#" + String(stepperStopIndex));
    }

    markCaptureDoneOnRobotDoc();

    captureWaitActive = false;
  }

  // Done waiting -> continue to next stop
  stepperPauseActive = false;

  if (movingForward) {
    if (nextStop < FINAL_TARGET) {
      nextStop += STEPS_PER_STOP;

      Serial.print("Next forward target: ");
      Serial.print(nextStop / MOTOR_STEPS_PER_REV);
      Serial.println(" rounds");

      logScanEvent("STEPPER_NEXT_FORWARD", currentPositionLabel + " next forward target " + String(nextStop / MOTOR_STEPS_PER_REV) + " rounds");

      stepper.moveTo(nextStop);
      return;
        } else {
  // Reached top, set servo to 40 before going down
  if (!nozzleServo.attached()) {
    nozzleServo.attach(SERVO_PIN, 500, 2400);
    delay(100);
  }
  nozzleServo.write(SERVO_ANGLE_A);   // 40 deg
  delay(500);

  movingForward = false;
  nextStop -= STEPS_PER_STOP;

  Serial.println("50 rounds reached. Servo set to 40. Switching to CCW.");
  logScanEvent("STEPPER_SWITCH_CCW", currentPositionLabel + " reached 50 rounds, servo=40, switching to CCW");

  stepper.moveTo(nextStop);
  return;
}
  } else {
    if (nextStop > 0) {
      nextStop -= STEPS_PER_STOP;

      Serial.print("Next backward target: ");
      Serial.print(nextStop / MOTOR_STEPS_PER_REV);
      Serial.println(" rounds");

      logScanEvent("STEPPER_NEXT_BACKWARD", currentPositionLabel + " next backward target " + String(nextStop / MOTOR_STEPS_PER_REV) + " rounds");

      stepper.moveTo(nextStop);
      return;
    } else {
      Serial.println("Stepper back at 0. Scan complete.");
      logScanEvent("STEPPER_BACK_AT_ZERO", currentPositionLabel + " stepper returned to 0");
      finalizePositionAfterScan();
      return;
    }
  }
}

void handlePositionStop() {
  plantStopCount++;
  currentPositionLabel = getPositionLabelFromCount(plantStopCount);

  stopMotors();
  robotMovingStatus = "STOPPED";
  scanningStatus = "WAITING";
  modeStr = "LINE_FOLLOW";
  pushDataToFirebase();

  Serial.println(">>> [STOP] POSITION " + currentPositionLabel + " DETECTED");
  Serial.println(">>> Waiting 5 seconds...");
  logScanEvent("POSITION_DETECTED", "Position " + currentPositionLabel + " detected, waiting 5s");

  delay(5000);

  robotMovingStatus = "STOPPED";
  scanningStatus = "SCANNING";
  pushDataToFirebase();

  startStepperScanSequence();
}

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  // === WIFI ===
  Serial.print("Connecting to WiFi");
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(300);
  }
  Serial.println("\nConnected to WiFi!");
  WiFi.setSleep(false);

  // TLS time
  initTime();

  // === FIREBASE INIT ===
  config.api_key = API_KEY;
  config.token_status_callback = tokenStatusCallback;

  if (Firebase.signUp(&config, &auth, "", "")) {
    Serial.println("Firebase Auth Ready");
  } else {
    Serial.print("Signup error: ");
    Serial.println(config.signer.signupError.message.c_str());
  }

  fbdo.setBSSLBufferSize(4096, 1024);
  fbdo.setResponseSize(4096);

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  // Pumps control pins
  pinMode(PUMP1_CTRL, OUTPUT);
  pinMode(PUMP2_CTRL, OUTPUT);
  pump1Off();
  pump2Off();

  // Servo
  nozzleServo.setPeriodHertz(50);

  // Motor Enable Pins (only if actually controlled by ESP32)
  if (USE_GPIO_EN_PINS) {
    pinMode(L_REN, OUTPUT);
    pinMode(L_LEN, OUTPUT);
    pinMode(R_REN, OUTPUT);
    pinMode(R_LEN, OUTPUT);

    digitalWrite(L_REN, HIGH);
    digitalWrite(L_LEN, HIGH);
    digitalWrite(R_REN, HIGH);
    digitalWrite(R_LEN, HIGH);
  }

  // PWM Setup
  ledcSetup(CH_L_RPWM, PWM_FREQ, PWM_RES);
  ledcSetup(CH_L_LPWM, PWM_FREQ, PWM_RES);
  ledcSetup(CH_R_RPWM, PWM_FREQ, PWM_RES);
  ledcSetup(CH_R_LPWM, PWM_FREQ, PWM_RES);

  ledcAttachPin(L_RPWM, CH_L_RPWM);
  ledcAttachPin(L_LPWM, CH_L_LPWM);
  ledcAttachPin(R_RPWM, CH_R_RPWM);
  ledcAttachPin(R_LPWM, CH_R_LPWM);

  // IR Sensors
  pinMode(IR_LEFT, INPUT);
  pinMode(IR_RIGHT, INPUT);

  // RFID
  SPI.begin();
  mfrc522.PCD_Init();
  Serial.println("RFID Ready");

  // Stepper enable
  pinMode(EN_PIN, OUTPUT);
  digitalWrite(EN_PIN, LOW);
  delay(100);

  stepper.setMaxSpeed(STEPPER_MAX_SPEED);
  stepper.setAcceleration(STEPPER_ACCEL);
  stepper.setCurrentPosition(0);

  stopMotors();

  // Load assigned tunnel once on boot
  fetchAssignedTunnelId();
  lastTunnelFetchMs = millis();

  pushDataToFirebase();
  logScanEvent("BOOT", "Robot booted and connected to WiFi/Firebase");

  Serial.println(">>> SYSTEM READY");
  Serial.println(">>> Robot starts with line following");
  Serial.println(">>> After RFID detected, next both-black = A, then B, then C");
}

// ================= LOOP =================
void loop() {
  if (Firebase.ready()) {
    unsigned long nowMs = millis();
    if (nowMs - lastTunnelFetchMs > TUNNEL_REFRESH_MS) {
      fetchAssignedTunnelId();
      lastTunnelFetchMs = nowMs;
    }
  }

  if (mode == FINISHED) {
    stopMotors();
    return;
  }

  if (mode == STEPPER_SCAN) {
    handleStepperScanMode();
    return;
  }

  if (mode == ACTUATION) {
    return;
  }

  // --------- MODE: LINE FOLLOW ---------
  if (mode == LINE_FOLLOW) {
    modeStr = "LINE_FOLLOW";

    int leftIR = digitalRead(IR_LEFT);
    int rightIR = digitalRead(IR_RIGHT);

    bool bothBlack = (leftIR == HIGH && rightIR == HIGH);

    // Debug IR states
    static unsigned long lastIrPrint = 0;
    if (millis() - lastIrPrint > 500) {
      Serial.print("IR Left: ");
      Serial.print(leftIR);
      Serial.print(" | IR Right: ");
      Serial.print(rightIR);
      Serial.print(" | bothBlack: ");
      Serial.println(bothBlack ? "YES" : "NO");
      lastIrPrint = millis();
    }

    // ===== RFID DETECTION =====
    if (!rfidDetected) {
      if (mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial()) {
        String readRFID = getRFIDTag();
        unsigned long nowMs = millis();

        if ((nowMs - lastRFIDTimeMs) >= RFID_COOLDOWN_MS || readRFID != lastRFID) {
          lastRFID = readRFID;
          lastRFIDTimeMs = nowMs;

          rfidDetected = true;
          plantStopCount = 0;
          currentPositionLabel = "RFID";
          currentRFID = readRFID;
          robotMovingStatus = "STOPPED";
          scanningStatus = "RFID_DETECTED";
          actuatorStatus = "OFF";

          stopMotors();
          pushDataToFirebase();
          updatePlantDocOnRFID();
          logScanEvent("RFID_DETECTED", "RFID detected. Next both-black will be position A");

          Serial.println(">>> [RFID] TAG DETECTED");
          Serial.println(">>> Next both-black will be POSITION A");

          delay(1000);
        }

        mfrc522.PICC_HaltA();
        mfrc522.PCD_StopCrypto1();
      }
    }

    // ===== POSITION DETECTION AFTER RFID =====
    if (rfidDetected) {
      if (bothBlack && !bothBlackLatched) {
        Serial.println(">>> bothBlack edge detected after RFID. Stopping for position.");
        bothBlackLatched = true;
        handlePositionStop();
        return;
      }

      if (!bothBlack) {
        bothBlackLatched = false;
      }
    }

    // ===== NORMAL LINE FOLLOWING =====
    if (leftIR == LOW && rightIR == HIGH) {
      leftTurn();
    }
    else if (leftIR == HIGH && rightIR == LOW) {
      rightTurn();
    }
    else {
      forward();
    }

    robotMovingStatus = "MOVING";
    scanningStatus = "IDLE";
    actuatorStatus = "OFF";

    delay(10);
    return;
  }
}
