String getVisualHealth(float t1, float h1, float t2, float h2) {
  bool t1Valid = isfinite(t1) && t1 >= TEMP_MIN_C && t1 <= TEMP_MAX_C;
  bool t2Valid = isfinite(t2) && t2 >= TEMP_MIN_C && t2 <= TEMP_MAX_C;
  bool h1Valid = isfinite(h1) && h1 >= HUM_MIN_PCT && h1 <= HUM_MAX_PCT;
  bool h2Valid = isfinite(h2) && h2 >= HUM_MIN_PCT && h2 <= HUM_MAX_PCT;

  if (!t1Valid || !t2Valid || !h1Valid || !h2Valid) {
    return "FAULT";
  }

  bool tempMismatch = fabs(t1 - t2) > TEMP_MISMATCH_C;
  bool humMismatch = fabs(h1 - h2) > HUM_MISMATCH_PCT;

  if (tempMismatch || humMismatch) {
    return "WARN";
  }

  return "OK";
}

// ================= TIME TLS =================
bool initTime() {
  configTime(19800, 0, "pool.ntp.org", "time.nist.gov"); // Sri Lanka UTC+5:30

  time_t now = time(nullptr);
  int retry = 0;

  while (now < 1700000000 && retry < 60) {
    delay(500);
    now = time(nullptr);
    retry++;
  }

  return now >= 1700000000;
}

String nowRFC3339Z() {
  time_t now = time(nullptr);
  if (now < 1700000000) return "1970-01-01T00:00:00Z";

  struct tm tm_utc;
  gmtime_r(&now, &tm_utc);

  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &tm_utc);

  return String(buf);
}

// ================= LORA =================
bool initLoRaWithOLED() {
  oledStatus("LoRa: Initializing...", "433MHz SX1278");

  SPI.begin(LORA_SCK, LORA_MISO, LORA_MOSI, LORA_CS);
  LoRa.setPins(LORA_CS, LORA_RST, LORA_DIO0);

  if (!LoRa.begin(LORA_FREQ)) {
    oledStatus("LoRa: INIT FAILED", "Check wiring", "3.3V + antenna");
    return false;
  }

  LoRa.setSpreadingFactor(7);
  LoRa.setSignalBandwidth(125E3);
  LoRa.setCodingRate4(5);

  oledStatus("LoRa: Receiver Ready");
  delay(700);

  return true;
}

// ================= PARSE =================
// incoming format: counter,t1,h1,t2,h2
bool parseCSV5(const String &s, String &counterStr, float &t1, float &h1, float &t2, float &h2) {
  int p1 = s.indexOf(',');
  int p2 = s.indexOf(',', p1 + 1);
  int p3 = s.indexOf(',', p2 + 1);
  int p4 = s.indexOf(',', p3 + 1);

  if (p1 < 0 || p2 < 0 || p3 < 0 || p4 < 0) return false;

  counterStr = s.substring(0, p1);
  t1 = s.substring(p1 + 1, p2).toFloat();
  h1 = s.substring(p2 + 1, p3).toFloat();
  t2 = s.substring(p3 + 1, p4).toFloat();
  h2 = s.substring(p4 + 1).toFloat();

  return true;
}

// ================= SENSOR VALIDATION + ANOMALY =================
bool validRange(float v, float lo, float hi) {
  return isfinite(v) && v >= lo && v <= hi;
}

void computeAnomaly(
  bool &s1TempValid,
  bool &s2TempValid,
  bool &s1HumValid,
  bool &s2HumValid,
  bool &tempMismatch,
  bool &humMismatch,
  float &tempDiff,
  float &humDiff,
  String &level,
  String &message
) {
  s1TempValid = validRange(lastT1, TEMP_MIN_C, TEMP_MAX_C);
  s2TempValid = validRange(lastT2, TEMP_MIN_C, TEMP_MAX_C);
  s1HumValid  = validRange(lastH1, HUM_MIN_PCT, HUM_MAX_PCT);
  s2HumValid  = validRange(lastH2, HUM_MIN_PCT, HUM_MAX_PCT);

  tempDiff = NAN;
  humDiff  = NAN;
  tempMismatch = false;
  humMismatch  = false;

  if (s1TempValid && s2TempValid) {
    tempDiff = fabs(lastT1 - lastT2);
    tempMismatch = tempDiff > TEMP_MISMATCH_C;
  }

  if (s1HumValid && s2HumValid) {
    humDiff = fabs(lastH1 - lastH2);
    humMismatch = humDiff > HUM_MISMATCH_PCT;
  }

  if (!s1TempValid || !s2TempValid || !s1HumValid || !s2HumValid) {
    level = "FAULT";
  } else if (tempMismatch || humMismatch) {
    level = "WARN";
  } else {
    level = "OK";
  }

  String msg = "";

  if (level == "OK") {
    msg = "All sensors OK";
  } else {
    if (!s1TempValid) msg += "T1 invalid; ";
    if (!s2TempValid) msg += "T2 invalid; ";
    if (!s1HumValid)  msg += "H1 invalid; ";
    if (!s2HumValid)  msg += "H2 invalid; ";
    if (tempMismatch) msg += "Temp mismatch; ";
    if (humMismatch)  msg += "Humidity mismatch; ";
  }

  msg.trim();
  message = msg;
}

String getFaultType(
  bool s1TempValid,
  bool s2TempValid,
  bool s1HumValid,
  bool s2HumValid,
  bool tempMismatch,
  bool humMismatch
) {
  if (!s1TempValid || !s2TempValid || !s1HumValid || !s2HumValid) {
    return "INVALID_SENSOR_READING";
  }

  if (tempMismatch) {
    return "TEMPERATURE_MISMATCH";
  }

  if (humMismatch) {
    return "HUMIDITY_MISMATCH";
  }

  return "SENSOR_WARNING";
}

// ================= ENSURE GATEWAY DOC AUTO CREATE =================
void ensureGatewayDoc() {
  if (!Firebase.ready()) return;

  String path = String(GATEWAYS_COLLECTION) + "/" + String(GATEWAY_ID);

  FirebaseJson content;
  content.set("fields/gatewayId/stringValue", String(GATEWAY_ID));
  content.set("fields/assignedTunnelId/stringValue", assignedTunnelId);
  content.set("fields/lastSeenAt/timestampValue", nowRFC3339Z());

  if (!Firebase.Firestore.createDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str(), content.raw())) {
    String reason = fbdo.errorReason();

    if (reason.indexOf("ALREADY_EXISTS") >= 0 || reason.indexOf("already exists") >= 0) {
      return;
    }

    Serial.print("ensureGatewayDoc FAILED: ");
    Serial.println(reason);
  }
}

// ================= FETCH ASSIGNED TUNNEL =================
void fetchAssignedTunnelId() {
  if (!Firebase.ready()) return;

  String path = String(GATEWAYS_COLLECTION) + "/" + String(GATEWAY_ID);

  if (!Firebase.Firestore.getDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str())) {
    return;
  }

  FirebaseJson json;
  json.setJsonData(fbdo.payload().c_str());

  FirebaseJsonData data;
  json.get(data, "fields/assignedTunnelId/stringValue");

  if (data.success) {
    assignedTunnelId = data.to<String>();
  }
}

// ================= UPDATE GATEWAY DOC =================
void updateGatewayDoc(const String &anLevel) {
  if (!Firebase.ready()) return;

  ensureGatewayDoc();

  String path = String(GATEWAYS_COLLECTION) + "/" + String(GATEWAY_ID);

  FirebaseJson content;
  content.set("fields/gatewayId/stringValue", String(GATEWAY_ID));
  content.set("fields/assignedTunnelId/stringValue", assignedTunnelId);
  content.set("fields/lastSeenAt/timestampValue", nowRFC3339Z());
  content.set("fields/lastRssi/integerValue", (int)lastRSSI);
  content.set("fields/lastSnr/doubleValue", isnan(lastSNR) ? 0.0 : lastSNR);
  content.set("fields/health/stringValue", anLevel);

  String mask = "gatewayId,assignedTunnelId,lastSeenAt,lastRssi,lastSnr,health";

  if (!Firebase.Firestore.patchDocument(&fbdo, FIREBASE_PROJECT_ID, "", path.c_str(), content.raw(), mask.c_str())) {
    Serial.print("Gateway patch FAILED: ");
    Serial.println(fbdo.errorReason());
  }
}

// ================= FAULT LOG WRITE =================
void maybeWriteSensorFaultLog(
  const String &anLevel,
  const String &anMessage,
  bool s1TempValid,
  bool s2TempValid,
  bool s1HumValid,
  bool s2HumValid,
  bool tempMismatch,
  bool humMismatch,
  float tempDiff,
  float humDiff
) {
  if (!Firebase.ready()) return;
  if (assignedTunnelId.length() == 0) return;

  if (anLevel == "OK") {
    lastFaultKey = "";
    return;
  }

  String faultType = getFaultType(
    s1TempValid,
    s2TempValid,
    s1HumValid,
    s2HumValid,
    tempMismatch,
    humMismatch
  );

  String faultKey = anLevel + "|" + faultType + "|" + anMessage;

  if (faultKey == lastFaultKey && millis() - lastFaultLogMs < FAULT_LOG_COOLDOWN_MS) {
    return;
  }

  String logPath =
    String("tunnels/") +
    assignedTunnelId +
    "/" +
    String(FAULT_LOGS_SUBCOLLECTION) +
    "/" +
    makeDocId();

  FirebaseJson log;

  log.set("fields/gatewayId/stringValue", String(GATEWAY_ID));
  log.set("fields/tunnelId/stringValue", assignedTunnelId);
  log.set("fields/type/stringValue", faultType);
  log.set("fields/level/stringValue", anLevel);
  log.set("fields/message/stringValue", anMessage);
  log.set("fields/status/stringValue", "ACTIVE");

  log.set("fields/s1_temp/doubleValue", lastT1);
  log.set("fields/s1_hum/doubleValue", lastH1);
  log.set("fields/s2_temp/doubleValue", lastT2);
  log.set("fields/s2_hum/doubleValue", lastH2);

  log.set("fields/s1_temp_valid/booleanValue", s1TempValid);
  log.set("fields/s2_temp_valid/booleanValue", s2TempValid);
  log.set("fields/s1_hum_valid/booleanValue", s1HumValid);
  log.set("fields/s2_hum_valid/booleanValue", s2HumValid);

  log.set("fields/tempMismatch/booleanValue", tempMismatch);
  log.set("fields/humMismatch/booleanValue", humMismatch);

  if (isfinite(tempDiff)) {
    log.set("fields/tempDiff/doubleValue", tempDiff);
  }

  if (isfinite(humDiff)) {
    log.set("fields/humDiff/doubleValue", humDiff);
  }

  log.set("fields/rssi/integerValue", (int)lastRSSI);
  log.set("fields/snr/doubleValue", isnan(lastSNR) ? 0.0 : lastSNR);
  log.set("fields/createdAt/timestampValue", nowRFC3339Z());

  bool ok = Firebase.Firestore.createDocument(
    &fbdo,
    FIREBASE_PROJECT_ID,
    "",
    logPath.c_str(),
    log.raw()
  );

  if (ok) {
    Serial.println("Sensor fault log created.");
    Serial.print("Fault Type: ");
    Serial.println(faultType);

    lastFaultKey = faultKey;
    lastFaultLogMs = millis();
  } else {
    Serial.print("Sensor fault log FAILED: ");
    Serial.println(fbdo.errorReason());
  }
}

void writeSensorTimeoutFaultIfNeeded() {
  if (!Firebase.ready()) return;
  if (assignedTunnelId.length() == 0) return;

  unsigned long nowMs = millis();

  if (!haveData) {
    return;
  }

  if (nowMs - lastPacketMs < SENSOR_TIMEOUT_FAULT_MS) {
    timeoutFaultActive = false;
    return;
  }

  if (timeoutFaultActive && nowMs - lastTimeoutFaultLogMs < FAULT_LOG_COOLDOWN_MS) {
    return;
  }

  String logPath =
    String("tunnels/") +
    assignedTunnelId +
    "/" +
    String(FAULT_LOGS_SUBCOLLECTION) +
    "/" +
    makeDocId();

  FirebaseJson log;

  log.set("fields/gatewayId/stringValue", String(GATEWAY_ID));
  log.set("fields/tunnelId/stringValue", assignedTunnelId);
  log.set("fields/type/stringValue", "SENSOR_TIMEOUT");
  log.set("fields/level/stringValue", "FAULT");
  log.set("fields/message/stringValue", "No LoRa sensor packet received for more than 60 seconds");
  log.set("fields/status/stringValue", "ACTIVE");

  log.set("fields/s1_temp/doubleValue", lastT1);
  log.set("fields/s1_hum/doubleValue", lastH1);
  log.set("fields/s2_temp/doubleValue", lastT2);
  log.set("fields/s2_hum/doubleValue", lastH2);

  log.set("fields/rssi/integerValue", (int)lastRSSI);
  log.set("fields/snr/doubleValue", isnan(lastSNR) ? 0.0 : lastSNR);
  log.set("fields/createdAt/timestampValue", nowRFC3339Z());

  bool ok = Firebase.Firestore.createDocument(
    &fbdo,
    FIREBASE_PROJECT_ID,
    "",
    logPath.c_str(),
    log.raw()
  );

  if (ok) {
    Serial.println("Sensor timeout fault log created.");

    timeoutFaultActive = true;
    lastTimeoutFaultLogMs = nowMs;

    updateGatewayDoc("FAULT");
  } else {
    Serial.print("Sensor timeout fault log FAILED: ");
    Serial.println(fbdo.errorReason());
  }
}

// ================= FIRESTORE WRITE =================
bool storeToFirestore() {
  if (!Firebase.ready()) return false;
  if (assignedTunnelId.length() == 0) return false;

  bool s1TempValid, s2TempValid, s1HumValid, s2HumValid;
  bool tempMismatch, humMismatch;
  float tempDiff, humDiff;
  String anLevel, anMessage;

  computeAnomaly(
    s1TempValid,
    s2TempValid,
    s1HumValid,
    s2HumValid,
    tempMismatch,
    humMismatch,
    tempDiff,
    humDiff,
    anLevel,
    anMessage
  );

  String readingPath =
    String("tunnels/") +
    assignedTunnelId +
    "/" +
    String(READINGS_SUBCOLLECTION) +
    "/" +
    makeDocId();

  String summaryPath =
    String("tunnels/") +
    assignedTunnelId +
    "/sensorSummary/latest";

  float avgT = (lastT1 + lastT2) / 2.0;
  float avgH = (lastH1 + lastH2) / 2.0;

  // 1. Create reading doc
  FirebaseJson reading;

  reading.set("fields/gatewayId/stringValue", String(GATEWAY_ID));
  reading.set("fields/counter/stringValue", lastCounter);

  reading.set("fields/s1_temp/doubleValue", lastT1);
  reading.set("fields/s1_hum/doubleValue", lastH1);
  reading.set("fields/s2_temp/doubleValue", lastT2);
  reading.set("fields/s2_hum/doubleValue", lastH2);

  reading.set("fields/avg_temp/doubleValue", avgT);
  reading.set("fields/avg_hum/doubleValue", avgH);

  reading.set("fields/rssi/integerValue", (int)lastRSSI);
  reading.set("fields/snr/doubleValue", isnan(lastSNR) ? 0.0 : lastSNR);
  reading.set("fields/ts/timestampValue", nowRFC3339Z());

  reading.set("fields/an_level/stringValue", anLevel);
  reading.set("fields/an_message/stringValue", anMessage);
  reading.set("fields/an_s1_temp_valid/booleanValue", s1TempValid);
  reading.set("fields/an_s2_temp_valid/booleanValue", s2TempValid);
  reading.set("fields/an_s1_hum_valid/booleanValue", s1HumValid);
  reading.set("fields/an_s2_hum_valid/booleanValue", s2HumValid);
  reading.set("fields/an_tempMismatch/booleanValue", tempMismatch);
  reading.set("fields/an_humMismatch/booleanValue", humMismatch);

  if (isfinite(tempDiff)) {
    reading.set("fields/an_tempDiff/doubleValue", tempDiff);
  }

  if (isfinite(humDiff)) {
    reading.set("fields/an_humDiff/doubleValue", humDiff);
  }

  bool ok1 = Firebase.Firestore.createDocument(
    &fbdo,
    FIREBASE_PROJECT_ID,
    "",
    readingPath.c_str(),
    reading.raw()
  );

  if (!ok1) {
    Serial.print("Reading create FAILED: ");
    Serial.println(fbdo.errorReason());
  }

  // 2. Patch summary/latest for fast mobile UI
  FirebaseJson summary;

  summary.set("fields/gatewayId/stringValue", String(GATEWAY_ID));
  summary.set("fields/counter/stringValue", lastCounter);

  summary.set("fields/s1_temp/doubleValue", lastT1);
  summary.set("fields/s2_temp/doubleValue", lastT2);
  summary.set("fields/avg_temp/doubleValue", avgT);

  summary.set("fields/s1_hum/doubleValue", lastH1);
  summary.set("fields/s2_hum/doubleValue", lastH2);
  summary.set("fields/avg_hum/doubleValue", avgH);

  summary.set("fields/rssi/integerValue", (int)lastRSSI);
  summary.set("fields/snr/doubleValue", isnan(lastSNR) ? 0.0 : lastSNR);
  summary.set("fields/lastSeenAt/timestampValue", nowRFC3339Z());

  summary.set("fields/an_level/stringValue", anLevel);
  summary.set("fields/an_message/stringValue", anMessage);
  summary.set("fields/an_s1_temp_valid/booleanValue", s1TempValid);
  summary.set("fields/an_s2_temp_valid/booleanValue", s2TempValid);
  summary.set("fields/an_s1_hum_valid/booleanValue", s1HumValid);
  summary.set("fields/an_s2_hum_valid/booleanValue", s2HumValid);
  summary.set("fields/an_tempMismatch/booleanValue", tempMismatch);
  summary.set("fields/an_humMismatch/booleanValue", humMismatch);

  if (isfinite(tempDiff)) {
    summary.set("fields/an_tempDiff/doubleValue", tempDiff);
  }

  if (isfinite(humDiff)) {
    summary.set("fields/an_humDiff/doubleValue", humDiff);
  }

  String mask =
    "gatewayId,counter,s1_temp,s2_temp,avg_temp,s1_hum,s2_hum,avg_hum,"
    "rssi,snr,lastSeenAt,"
    "an_level,an_message,an_s1_temp_valid,an_s2_temp_valid,"
    "an_s1_hum_valid,an_s2_hum_valid,an_tempMismatch,an_humMismatch,"
    "an_tempDiff,an_humDiff";

  bool ok2 = Firebase.Firestore.patchDocument(
    &fbdo,
    FIREBASE_PROJECT_ID,
    "",
    summaryPath.c_str(),
    summary.raw(),
    mask.c_str()
  );

  if (!ok2) {
    Serial.print("Summary patch FAILED: ");
    Serial.println(fbdo.errorReason());
  }

  updateGatewayDoc(anLevel);

  maybeWriteSensorFaultLog(
    anLevel,
    anMessage,
    s1TempValid,
    s2TempValid,
    s1HumValid,
    s2HumValid,
    tempMismatch,
    humMismatch,
    tempDiff,
    humDiff
  );

  return ok1 && ok2;
}

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  Wire.begin(I2C_SDA, I2C_SCL);

  if (!oledBegin()) {
    Serial.println("OLED not found at 0x3C or 0x3D");
    while (1) delay(10);
  }

  oledStatus("Booting...");
  delay(500);

  connectWiFiWithOLED();

  oledStatus("Time: Syncing NTP...");
  bool timeOk = initTime();
  oledStatus(String("Time: ") + (timeOk ? "OK" : "FAILED"));
  delay(700);

  initFirebaseWithOLED();

  if (!initLoRaWithOLED()) {
    while (1) delay(10);
  }

  // Create gateway doc so loraGateways appears
  fetchAssignedTunnelId();
  ensureGatewayDoc();
  updateGatewayDoc("OK");

  lastAssignFetchMs = millis();
  lastEnsureMs = millis();

  lastPacketMs = millis();
  showWaitingScreen();

  Serial.println("Receiver ready.");
}

// ================= LOOP =================
void loop() {
  Firebase.ready(); // keep token alive

  // periodic ensure
  if (Firebase.ready() && (millis() - lastEnsureMs > ENSURE_INTERVAL_MS)) {
    ensureGatewayDoc();
    lastEnsureMs = millis();
  }

  // refresh assigned tunnel
  if (Firebase.ready() && (millis() - lastAssignFetchMs > ASSIGN_REFRESH_MS)) {
    fetchAssignedTunnelId();
    lastAssignFetchMs = millis();
  }

  int packetSize = LoRa.parsePacket();

  if (packetSize) {
    String incoming = "";

    while (LoRa.available()) {
      incoming += (char)LoRa.read();
    }

    serialLine();
    Serial.println("LORA PACKET RECEIVED");
    Serial.print("Raw Packet: ");
    Serial.println(incoming);
    serialLine();

    int rssi = LoRa.packetRssi();
    float snr = LoRa.packetSnr();

    String counterStr;
    float t1, h1, t2, h2;

    if (parseCSV5(incoming, counterStr, t1, h1, t2, h2)) {
      lastPacketMs = millis();
      timeoutFaultActive = false;

      lastCounter = counterStr;
      lastT1 = t1;
      lastH1 = h1;
      lastT2 = t2;
      lastH2 = h2;
      lastRSSI = rssi;
      lastSNR = snr;
      haveData = true;

      drawData(t1, h1, t2, h2, rssi, snr);
    } else {
      Serial.println("Invalid LoRa CSV packet format.");
    }
  }

  if ((millis() - lastPacketMs) > WAITING_TIMEOUT_MS) {
    if (!showingWaiting) {
      showWaitingScreen();
    }
  }

  // Real downtime fault log after 60 seconds without LoRa data
  writeSensorTimeoutFaultIfNeeded();

  if (haveData && (millis() - lastStoreMs >= STORE_INTERVAL_MS)) {
    lastStoreMs = millis();

    if (WiFi.status() != WL_CONNECTED) {
      oledStatus("WiFi: Reconnecting...");
      connectWiFiWithOLED();

      oledStatus("Time: Syncing NTP...");
      initTime();

      fetchAssignedTunnelId();
      showWaitingScreen();
    }

    bool firestoreSaved = storeToFirestore();

    Serial.print("Firestore Save: ");
    Serial.println(firestoreSaved ? "SUCCESS" : "FAILED / SKIPPED");
  }
}
